import { db } from "../js/firebase.js"; import { collection, addDoc, getDocs, doc, deleteDoc, serverTimestamp, query, orderBy } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-firestore.js"; // Load existing notices on page load loadNotices(); // Publish Notice Button Event document.getElementById("publishNoticeBtn").addEventListener("click", async () => { const title = document.getElementById("noticeTitle").value.trim(); const message = document.getElementById("noticeMessage").value.trim(); if (!title || !message) { alert("Please fill in both title and message."); return; } try { await addDoc(collection(db, "notifications"), { title: title, message: message, createdAt: serverTimestamp() }); alert("Notice Published Successfully!"); document.getElementById("noticeTitle").value = ""; document.getElementById("noticeMessage").value = ""; loadNotices(); } catch (error) { console.error("Error publishing notice: ", error); alert("Failed to publish notice."); } }); // Function to load and display notices async function loadNotices() { const noticeList = document.getElementById("noticeList"); noticeList.innerHTML = "Loading..."; try { const q = query(collection(db, "notifications"), orderBy("createdAt", "desc")); const querySnapshot = await getDocs(q); let html = ""; if (querySnapshot.empty) { noticeList.innerHTML = "

No notices found.

"; return; } querySnapshot.forEach((documentSnapshot) => { const data = documentSnapshot.data(); const id = documentSnapshot.id; html += `

${data.title}

${data.message}

`; }); noticeList.innerHTML = html; } catch (error) { console.error("Error loading notices: ", error); noticeList.innerHTML = "

Failed to load notices.

"; } } // Function to delete a notice window.deleteNotice = async function(id) { if (confirm("Are you sure you want to delete this notice?")) { try { await deleteDoc(doc(db, "notifications", id)); alert("Notice Deleted"); loadNotices(); } catch (error) { console.error("Error deleting notice: ", error); alert("Failed to delete notice."); } } };
